function generateQuantumEntropy() {
  const sources = [
    crypto.randomBytes(32),
    Buffer.from(Date.now().toString()),
    Buffer.from(process.memoryUsage().heapUsed.toString())
  ];
  return crypto.createHash('sha256').update(Buffer.concat(sources)).digest();
}
